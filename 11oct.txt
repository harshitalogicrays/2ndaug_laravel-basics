    public function forcedelete($id){
        $customer = Customer::withTrashed()->find($id);
        if($customer) {$customer->forcedelete();}
        return redirect()->back();
    }

    public function restore($id){
        $customer = Customer::withTrashed()->find($id);
        if($customer) {$customer->restore();}
        return redirect('/customer');
    }
